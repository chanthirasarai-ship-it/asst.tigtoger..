    <div class="flex-1 overflow-y-auto px-4 py-4 text-sm leading-relaxed text-gray-700 dark:text-gray-300 space-y-4" id="terms-body">
     <p class="font-bold text-base text-gray-900 dark:text-gray-100">ข้อตกลงและเงื่อนไขการใช้บริการแอปพลิเคชัน Asst.tigtoger</p>
     <p>ยินดีต้อนรับสู่ Asst.tigtoger ซึ่งให้บริการโดย UBSMKT. กรุณาอ่านข้อตกลงและเงื่อนไขการใช้บริการนี้อย่างละเอียดก่อนเริ่มต้นใช้งาน การลงทะเบียน เข้าถึง หรือใช้งานแอปพลิเคชันจะถือว่าท่านได้อ่าน เข้าใจ และยอมรับที่จะผูกพันตามข้อตกลงนี้ทุกประการ</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">1. การสร้างบัญชีและการรักษาความปลอดภัย</p>
     <p>1.1 ผู้ใช้บริการต้องให้ข้อมูลที่เป็นจริง ถูกต้อง และเป็นปัจจุบันในการลงทะเบียนเข้าใช้งาน</p>
     <p>1.2 ผู้ใช้บริการมีหน้าที่รับผิดชอบในการเก็บรักษาชื่อผู้ใช้ (Username) และรหัสผ่าน (Password) เป็นความลับ</p>
     <p>1.3 ผู้ใช้บริการต้องรับผิดชอบต่อกิจกรรมทั้งหมดที่เกิดขึ้นภายใต้บัญชีของตนเอง หากพบการเข้าถึงโดยไม่ได้รับอนุญาต ต้องแจ้งผู้ให้บริการทันที</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">2. ขอบเขตการให้บริการและการใช้งานที่เหมาะสม</p>
     <p>2.1 แอปพลิเคชันนี้ให้บริการเพื่อช่วยวางแผน สร้าง และตรวจสอบคอนเทนต์สำหรับผู้สร้างคอนเทนต์ โดยผู้ใช้ต้องตรวจสอบเนื้อหาและปฏิบัติตามกฎหมายและนโยบายของแพลตฟอร์มที่เกี่ยวข้อง</p>
     <p>2.2 ผู้ใช้บริการตกลงว่าจะไม่ใช้แอปพลิเคชันเพื่อวัตถุประสงค์ที่ผิดกฎหมาย ละเมิดสิทธิผู้อื่น หรือละเมิดศีลธรรมอันดี; ไม่พยายามดัดแปลง ย้อนกระบวนการสร้าง (Reverse Engineer) หรือส่งไวรัส/รหัสที่เป็นอันตรายเข้ามาในระบบ; และไม่คัดลอก ดัดแปลง หรือนำส่วนหนึ่งส่วนใดของแอปพลิเคชันไปใช้ในเชิงพาณิชย์โดยไม่ได้รับอนุญาต</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">3. ทรัพย์สินทางปัญญา</p>
     <p>โลโก้ เครื่องหมายการค้า ข้อความ รูปภาพ ซอฟต์แวร์ และเนื้อหาทั้งหมดภายในแอปพลิเคชัน ถือเป็นทรัพย์สินทางปัญญาของผู้ให้บริการ หรือผู้ให้อนุญาตแก่ผู้ให้บริการ ห้ามมิให้ผู้ใดคัดลอก ทำซ้ำ หรือนำไปเผยแพร่โดยไม่ได้รับยินยอมเป็นลายลักษณ์อักษร</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">4. รูปแบบสมาชิก การชำระเงิน และการต่ออายุ</p>
     <p>มีสมาชิกรายวัน 24 ชั่วโมง, รายสัปดาห์ 7 วัน, รายเดือน 30 วัน, รายปี 365 วัน และสมาชิกแบบถาวร/ซื้อขาด โดยการซื้อขาดชำระเงินครั้งเดียวเพื่อเข้าถึงบริการตลอดระยะเวลาที่แอปพลิเคชันยังคงเปิดให้บริการตามวงจรชีวิตของผลิตภัณฑ์</p>
     <p>ผู้ให้บริการจะแสดงรายละเอียดราคา รอบบิล และเงื่อนไขการต่ออายุอย่างชัดเจนก่อนสั่งซื้อ สำหรับแพ็กเกจรายวัน รายสัปดาห์ รายเดือน และรายปี ระบบอาจต่ออายุและตัดเงินอัตโนมัติเมื่อครบกำหนด เว้นแต่ผู้ใช้ยกเลิกก่อนรอบถัดไป ช่องทางชำระเงินอาจรวมถึง Apple App Store, Google Play Store หรือระบบชำระเงินภายนอกที่ได้รับอนุญาต</p>
     <p>ผู้ใช้ยกเลิกการต่ออายุอัตโนมัติได้ผ่านเมนูตั้งค่าในแอปหรือผ่านสโตร์ที่สมัคร อย่างน้อย 24 ชั่วโมงก่อนสิ้นสุดระยะเวลาปัจจุบัน เมื่อยกเลิกแล้วใช้สิทธิ์ได้จนหมดรอบและจะไม่ถูกเรียกเก็บในรอบถัดไป</p>
     <p>ค่าบริการที่ชำระแล้ว รวมถึงแพ็กเกจถาวร ไม่สามารถขอคืนเงิน แลกเปลี่ยน หรือเปลี่ยนเป็นเงินสดได้ เว้นแต่เป็นไปตามกฎหมายหรือนโยบายของสโตร์ที่เกี่ยวข้อง กรณีสมัครผ่าน App Store หรือ Google Play Store ให้ขอคืนเงินตามเงื่อนไขของ Apple หรือ Google โดยตรง</p>
     <p>ผู้ให้บริการอาจปรับเปลี่ยนราคา โดยจะแจ้งล่วงหน้าอย่างน้อย 30 วันผ่านแอปหรืออีเมลที่ลงทะเบียน หากไม่ยอมรับราคาใหม่ ผู้ใช้ยกเลิกก่อนวันที่มีผลได้</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">5. ข้อจำกัดความรับผิดชอบ</p>
     <p>ผู้ให้บริการมุ่งมั่นพัฒนาระบบให้มีประสิทธิภาพ แต่ไม่รับประกันว่าการทำงานจะไม่มีข้อผิดพลาดหรือหยุดชะงัก และไม่รับผิดชอบต่อความเสียหายทางตรง ทางอ้อม หรือผลพวงที่เกิดจากการใช้งานหรือไม่สามารถใช้งานแอปพลิเคชันได้ ทั้งนี้ต้องอยู่ภายใต้ขอบเขตที่กฎหมายอนุญาต</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">6. การระงับและการยกเลิกการใช้งาน</p>
     <p>ผู้ให้บริการขอสงวนสิทธิ์ระงับหรือยกเลิกบัญชีทันทีโดยไม่ต้องแจ้งล่วงหน้า หากพบการละเมิดข้อตกลงหรือการกระทำที่อาจก่อให้เกิดความเสียหายต่อระบบและผู้ใช้อื่น</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">7. การแก้ไขปรับปรุงข้อตกลง</p>
     <p>ผู้ให้บริการอาจแก้ไขหรือเปลี่ยนแปลงเงื่อนไขได้ โดยจะแจ้งผ่านแอปพลิเคชัน การใช้งานต่อหลังการเปลี่ยนแปลงถือว่ายอมรับข้อตกลงที่แก้ไข</p>
     <p class="font-semibold text-gray-900 dark:text-gray-100">8. กฎหมายที่ใช้บังคับ</p>
     <p>ข้อตกลงนี้อยู่ภายใต้กฎหมายของประเทศไทย</p>

// Packages
async function fetchBackendJson(url, options={}, timeoutMs=8000){
  const controller=new AbortController();
  const timer=setTimeout(()=>controller.abort(),timeoutMs);
  try{
    const response=await fetch(url,{...options,credentials:'same-origin',signal:controller.signal,headers:{Accept:'application/json',...(options.headers||{})}});
    let payload=null;
    try{payload=await response.json();}catch(_){payload=null;}
    if(!response.ok)throw new Error('backend_error');
    return payload;
  }finally{clearTimeout(timer);}
}

  }).catch(()=>{dataEl.innerHTML='<p class="text-sm text-gray-500 dark:text-gray-400">ตรวจสอบข้อมูลไม่ได้</p>';statusEl.innerHTML='<i data-lucide="alert-triangle" class="w-3.5 h-3.5 mt-0.5 shrink-0"></i> ยังไม่ได้เชื่อมต่อ Backend';lucide.createIcons();});
}
function initSystemStatus(){
  const checks=[['TikTok OAuth','/api/auth/tiktok/status'],['Backend/API','/api/health'],['ฐานข้อมูล','/api/health/database'],['Payment Gateway','/api/payments/status'],['Webhook','/api/payments/webhook/status'],['HTTPS','/api/health/https']];
  const list=document.getElementById('system-status-list');
  list.innerHTML=checks.map(([label])=>`<div class="flex items-center justify-between gap-2 rounded-lg bg-white dark:bg-slate-800 px-2 py-2"><span>${label}</span><span class="font-semibold text-gray-500">กำลังตรวจสอบ...</span></div>`).join('');
  checks.forEach(([label,url],i)=>fetchBackendJson(url).then(s=>{const item=list.children[i];if(item)item.querySelector('span:last-child').textContent=url==='/api/health'?'เชื่อมต่อ Backend แล้ว':(s?.ready?'พร้อม':'ยังไม่ได้ตั้งค่า Backend');}).catch(()=>{const item=list.children[i];if(item)item.querySelector('span:last-child').textContent='เชื่อมต่อ Backend ไม่สำเร็จ';}));
}
